#include "kernel.h"

#include <ATen/cuda/CUDAContext.h>
#include <c10/util/BFloat16.h>
#include <cuda_runtime.h>
#include <torch/extension.h>

#include <cmath>
#include <cstdint>

namespace {

constexpr int kHeadDim = 128;
constexpr int kBlockQ = 2;

__global__ void dense_qkv_prefill_causal_h8_kv1or8_d128_kernel(
    const c10::BFloat16* __restrict__ q,
    const c10::BFloat16* __restrict__ k,
    const c10::BFloat16* __restrict__ v,
    c10::BFloat16* __restrict__ out,
    int batch_size,
    int num_q_heads,
    int num_kv_heads,
    int seq_len,
    float sm_scale) {
  const int d = threadIdx.x;        // [0, 127]
  const int q_local = threadIdx.y;  // [0, 1]

  const int b = blockIdx.z;
  const int h = blockIdx.y;
  const int q_start = blockIdx.x * kBlockQ;
  const int qi = q_start + q_local;

  if (d >= kHeadDim || b >= batch_size || h >= num_q_heads) {
    return;
  }

  const bool valid_q = (qi < seq_len);
  const int kh = (num_kv_heads == 1) ? 0 : h;
  const int tile_q_end = q_start + kBlockQ - 1;
  const int max_k = (seq_len - 1 < tile_q_end) ? (seq_len - 1) : tile_q_end;

  __shared__ float k_tile[kHeadDim];
  __shared__ float v_tile[kHeadDim];
  __shared__ float red[kBlockQ][kHeadDim];
  __shared__ float sh_m[kBlockQ];
  __shared__ float sh_l[kBlockQ];
  __shared__ float sh_alpha[kBlockQ];
  __shared__ float sh_beta[kBlockQ];

  const int64_t q_head_base =
      (static_cast<int64_t>(b) * num_q_heads + h) * static_cast<int64_t>(seq_len) * kHeadDim;
  const int64_t kv_head_base =
      (static_cast<int64_t>(b) * num_kv_heads + kh) * static_cast<int64_t>(seq_len) * kHeadDim;

  float q_val = 0.0f;
  if (valid_q) {
    const int64_t q_idx = q_head_base + static_cast<int64_t>(qi) * kHeadDim + d;
    q_val = static_cast<float>(q[q_idx]);
  }

  if (d == 0) {
    sh_m[q_local] = -INFINITY;
    sh_l[q_local] = 0.0f;
    sh_alpha[q_local] = 1.0f;
    sh_beta[q_local] = 0.0f;
  }
  __syncthreads();

  float acc = 0.0f;

  if (max_k >= 0) {
    for (int kj = 0; kj <= max_k; ++kj) {
      if (q_local == 0) {
        const int64_t kv_idx = kv_head_base + static_cast<int64_t>(kj) * kHeadDim + d;
        k_tile[d] = static_cast<float>(k[kv_idx]);
        v_tile[d] = static_cast<float>(v[kv_idx]);
      }
      __syncthreads();

      const bool active = valid_q && (kj <= qi);
      red[q_local][d] = active ? (q_val * k_tile[d]) : 0.0f;
      __syncthreads();

      for (int stride = kHeadDim / 2; stride > 0; stride >>= 1) {
        if (d < stride) {
          red[q_local][d] += red[q_local][d + stride];
        }
        __syncthreads();
      }

      if (d == 0) {
        if (active) {
          const float score = red[q_local][0] * sm_scale;
          const float m_old = sh_m[q_local];
          const float l_old = sh_l[q_local];
          const float m_new = fmaxf(m_old, score);
          const float alpha = isfinite(m_old) ? expf(m_old - m_new) : 0.0f;
          const float beta = expf(score - m_new);
          sh_m[q_local] = m_new;
          sh_l[q_local] = l_old * alpha + beta;
          sh_alpha[q_local] = alpha;
          sh_beta[q_local] = beta;
        } else {
          sh_alpha[q_local] = 1.0f;
          sh_beta[q_local] = 0.0f;
        }
      }
      __syncthreads();

      if (valid_q) {
        const float alpha = sh_alpha[q_local];
        const float beta = sh_beta[q_local];
        acc = acc * alpha + beta * v_tile[d];
      }
      __syncthreads();
    }
  }

  if (valid_q) {
    const float denom = sh_l[q_local];
    const float out_val = (denom > 0.0f) ? (acc / denom) : 0.0f;
    const int64_t out_idx = q_head_base + static_cast<int64_t>(qi) * kHeadDim + d;
    out[out_idx] = c10::BFloat16(out_val);
  }
}

}  // namespace

void launch_dense_qkv_prefill_causal_h8_kv1or8_d128(
    const at::Tensor& q,
    const at::Tensor& k,
    const at::Tensor& v,
    float sm_scale,
    at::Tensor& out) {
  TORCH_CHECK(q.is_cuda(), "q must be on CUDA/ROCm device");
  TORCH_CHECK(k.is_cuda(), "k must be on CUDA/ROCm device");
  TORCH_CHECK(v.is_cuda(), "v must be on CUDA/ROCm device");
  TORCH_CHECK(out.is_cuda(), "out must be on CUDA/ROCm device");

  TORCH_CHECK(q.scalar_type() == at::kBFloat16, "q must be bfloat16");
  TORCH_CHECK(k.scalar_type() == at::kBFloat16, "k must be bfloat16");
  TORCH_CHECK(v.scalar_type() == at::kBFloat16, "v must be bfloat16");
  TORCH_CHECK(out.scalar_type() == at::kBFloat16, "out must be bfloat16");

  TORCH_CHECK(q.is_contiguous(), "q must be contiguous");
  TORCH_CHECK(k.is_contiguous(), "k must be contiguous");
  TORCH_CHECK(v.is_contiguous(), "v must be contiguous");
  TORCH_CHECK(out.is_contiguous(), "out must be contiguous");

  const int batch_size = static_cast<int>(q.size(0));
  const int num_q_heads = static_cast<int>(q.size(1));
  const int seq_len = static_cast<int>(q.size(2));
  const int head_dim = static_cast<int>(q.size(3));
  const int num_kv_heads = static_cast<int>(k.size(1));

  TORCH_CHECK(head_dim == kHeadDim, "head_dim must be 128");

  if (batch_size == 0 || seq_len == 0) {
    return;
  }

  const auto* q_ptr = q.data_ptr<c10::BFloat16>();
  const auto* k_ptr = k.data_ptr<c10::BFloat16>();
  const auto* v_ptr = v.data_ptr<c10::BFloat16>();
  auto* out_ptr = out.data_ptr<c10::BFloat16>();

  dim3 block(kHeadDim, kBlockQ, 1);
  dim3 grid((seq_len + kBlockQ - 1) / kBlockQ, num_q_heads, batch_size);

  cudaStream_t stream = at::cuda::getCurrentCUDAStream().stream();
  dense_qkv_prefill_causal_h8_kv1or8_d128_kernel<<<grid, block, 0, stream>>>(
      q_ptr, k_ptr, v_ptr, out_ptr, batch_size, num_q_heads, num_kv_heads, seq_len, sm_scale);

  cudaError_t err = cudaGetLastError();
  TORCH_CHECK(err == cudaSuccess, "Kernel launch failed: ", cudaGetErrorString(err));
}

<cpp_file name="main.cpp">
#include "kernel.h"

#include <ATen/cuda/CUDAGuard.h>
#include <torch/extension.h>

namespace py = pybind11;

torch::Tensor run(torch::Tensor q, torch::Tensor k, torch::Tensor v, double sm_scale) {
  at::NoGradGuard no_grad;

  TORCH_CHECK(q.dim() == 4, "q must be 4D [B, Hq, S, D]");
  TORCH_CHECK(k.dim() == 4, "k must be 4D [B, Hkv, S, D]");
  TORCH_CHECK(v.dim() == 4, "v must be 4D [B, Hkv, S, D]");

  const int64_t bq = q.size(0), hq = q.size(1), sq = q.size(2), dq = q.size(3);
  const int64_t bk = k.size(0), hk = k.size(1), sk = k.size(2), dk = k.size(3);
  const int64_t bv = v.size(0), hv = v.size(1), sv = v.size(2), dv = v.size(3);

  TORCH_CHECK(bq == bk && bq == bv, "Batch size mismatch across q/k/v");
  TORCH_CHECK(sq == sk && sq == sv, "Sequence length mismatch across q/k/v");
  TORCH_CHECK(dq == dk && dq == dv, "Head dim mismatch across q/k/v");

  TORCH_CHECK(hq == 8, "num_q_heads must be 8");
  TORCH_CHECK(hk == 1 || hk == 8, "num_kv_heads must be 1 or 8");
  TORCH_CHECK(hv == hk, "k and v num_kv_heads must match");
  TORCH_CHECK(dq == 128, "head_dim must be 128");

  torch::Device device = q.is_cuda() ? q.device() : torch::Device(torch::kCUDA, 0);
  at::cuda::CUDAGuard device_guard(device);

  auto qd = q.to(device, torch::kBFloat16, /*non_blocking=*/false, /*copy=*/false).contiguous();
  auto kd = k.to(device, torch::kBFloat16, /*non_blocking=*/false, /*copy=*/false).contiguous();
  auto vd = v.to(device, torch::kBFloat16, /*non_blocking=*/false, /*copy=*/false).contiguous();

  auto out = torch::empty({bq, hq, sq, dq}, qd.options().dtype(torch::kBFloat16));

  launch_dense_qkv_prefill_causal_h8_kv1or8_d128(qd, kd, vd, static_cast<float>(sm_scale), out);

  return out.to(torch::kCPU);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
  m.def(
      "run",
      &run,
      py::arg("q"),
      py::arg("k"),
      py::arg("v"),
      py::arg("sm_scale"),
      "Dense causal prefill attention (Hq=8, Hkv in {1,8}, D=128)");
}
