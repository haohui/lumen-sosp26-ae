import torch


@torch.no_grad()
def run(q, k, v, sm_scale):
    # dense SDPA reference (forward-only, causal prefill)
    assert q.ndim == 4 and k.ndim == 4 and v.ndim == 4
    bq, hq, sq, dq = q.shape
    bk, hk, sk, dk = k.shape
    bv, hv, sv, dv = v.shape

    assert (bq, sq, dq) == (bk, sk, dk) == (bv, sv, dv)
    assert hq == 8
    assert hk in (1, 8)
    assert hv == hk
    assert dq == 128

    if hk == 1:
        k = k.repeat_interleave(8, dim=1)
        v = v.repeat_interleave(8, dim=1)

    q = q.to(torch.float32)
    k = k.to(torch.float32)
    v = v.to(torch.float32)

    out = torch.nn.functional.scaled_dot_product_attention(
        q,
        k,
        v,
        attn_mask=None,
        dropout_p=0.0,
        is_causal=True,
        scale=float(sm_scale),
    )
    return out.to(torch.bfloat16)
