import torch
import triton
import triton.language as tl


@triton.jit
def _fused_causal_attention_mfma(
    q_ptr, k_ptr, v_ptr, o_ptr,
    B, HQ, HKV, S, D,
    stride_qb, stride_qh, stride_qs, stride_qd,
    stride_kb, stride_kh, stride_ks, stride_kd,
    stride_vb, stride_vh, stride_vs, stride_vd,
    stride_ob, stride_oh, stride_os, stride_od,
    scale,
    BLOCK_M: tl.constexpr,
    BLOCK_N: tl.constexpr,
    BLOCK_D: tl.constexpr,
):
    pid_m = tl.program_id(axis=0)
    pid_bh = tl.program_id(axis=1)

    b = pid_bh // HQ
    hq = pid_bh % HQ
    hkv = tl.where(HKV == 1, 0, hq)

    offs_m = pid_m * BLOCK_M + tl.arange(0, BLOCK_M)
    offs_d = tl.arange(0, BLOCK_D)

    q_ptrs = (
        q_ptr
        + b * stride_qb
        + hq * stride_qh
        + offs_m[:, None] * stride_qs
        + offs_d[None, :] * stride_qd
    )
    q_mask = (offs_m[:, None] < S) & (offs_d[None, :] < D)
    q = tl.load(q_ptrs, mask=q_mask, other=0).to(tl.bfloat16)

    m_i = tl.full((BLOCK_M,), -float("inf"), dtype=tl.float32)
    l_i = tl.zeros((BLOCK_M,), dtype=tl.float32)
    acc = tl.zeros((BLOCK_M, BLOCK_D), dtype=tl.float32)

    n_start = 0
    while n_start < S:
        offs_n = n_start + tl.arange(0, BLOCK_N)

        k_ptrs = (
            k_ptr
            + b * stride_kb
            + hkv * stride_kh
            + offs_n[:, None] * stride_ks
            + offs_d[None, :] * stride_kd
        )
        k_mask = (offs_n[:, None] < S) & (offs_d[None, :] < D)
        k = tl.load(k_ptrs, mask=k_mask, other=0).to(tl.bfloat16)

        qk = tl.dot(q, tl.trans(k), input_precision="ieee") * scale

        causal = offs_m[:, None] >= offs_n[None, :]
        valid_n = offs_n[None, :] < S
        valid_m = offs_m[:, None] < S
        qk = tl.where(causal & valid_n & valid_m, qk, -float("inf"))

        m_ij = tl.maximum(m_i, tl.max(qk, axis=1))
        p = tl.exp(qk - m_ij[:, None])
        l_ij = tl.sum(p, axis=1)

        alpha = tl.exp(m_i - m_ij)
        l_i = l_i * alpha + l_ij
        acc = acc * alpha[:, None]

        v_ptrs = (
            v_ptr
            + b * stride_vb
            + hkv * stride_vh
            + offs_n[:, None] * stride_vs
            + offs_d[None, :] * stride_vd
        )
        v_mask = (offs_n[:, None] < S) & (offs_d[None, :] < D)
        v = tl.load(v_ptrs, mask=v_mask, other=0).to(tl.bfloat16)

        acc = tl.dot(p.to(tl.bfloat16), v, acc, input_precision="ieee")
        m_i = m_ij
        n_start += BLOCK_N

    out = acc / l_i[:, None]

    o_ptrs = (
        o_ptr
        + b * stride_ob
        + hq * stride_oh
        + offs_m[:, None] * stride_os
        + offs_d[None, :] * stride_od
    )
    o_mask = (offs_m[:, None] < S) & (offs_d[None, :] < D)
    tl.store(o_ptrs, out.to(o_ptr.dtype.element_ty), mask=o_mask)

    BLOCK_N = 64
    BLOCK_D = 128
