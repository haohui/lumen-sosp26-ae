import torch
import torch.nn as nn
from torch.utils.cpp_extension import load_inline

_cpp_src = r"""
torch::Tensor bf16_matmul_hip(torch::Tensor A, torch::Tensor B);
"""

_hip_src = r"""
#include <torch/extension.h>
#include <hip/hip_runtime.h>
#include <hip/hip_bfloat16.h>

constexpr int TILE = 16;

__device__ __forceinline__ __hip_bfloat16 bf16_zero() {
    return __float2bfloat16(0.0f);
}

__global__ void bf16_gemm_tiled_kernel(
    const __hip_bfloat16* __restrict__ A,
    const __hip_bfloat16* __restrict__ B,
    __hip_bfloat16* __restrict__ C,
    int N, int K, int M
) {
    __shared__ __hip_bfloat16 As[TILE][TILE];
    __shared__ __hip_bfloat16 Bs[TILE][TILE];

    const int tx = threadIdx.x;
    const int ty = threadIdx.y;
    const int row = blockIdx.y * TILE + ty;
    const int col = blockIdx.x * TILE + tx;

    __hip_bfloat16 acc = bf16_zero();

    for (int t = 0; t < K; t += TILE) {
        const int a_col = t + tx;
        const int b_row = t + ty;

        As[ty][tx] = (row < N && a_col < K) ? A[row * K + a_col] : bf16_zero();
        Bs[ty][tx] = (b_row < K && col < M) ? B[b_row * M + col] : bf16_zero();

        __syncthreads();

        #pragma unroll
        for (int k = 0; k < TILE; ++k) {
            acc = acc + (As[ty][k] * Bs[k][tx]);
        }

        __syncthreads();
    }

    if (row < N && col < M) {
        C[row * M + col] = acc;
    }
}

torch::Tensor bf16_matmul_hip(torch::Tensor A, torch::Tensor B) {
    TORCH_CHECK(A.is_cuda(), "A must be a CUDA/HIP tensor");
    TORCH_CHECK(B.is_cuda(), "B must be a CUDA/HIP tensor");
    TORCH_CHECK(A.scalar_type() == at::kBFloat16, "A must be torch.bfloat16");
    TORCH_CHECK(B.scalar_type() == at::kBFloat16, "B must be torch.bfloat16");
    TORCH_CHECK(A.dim() == 2 && B.dim() == 2, "A and B must be 2D tensors");
    TORCH_CHECK(A.size(1) == B.size(0), "matmul dimension mismatch");

    auto A_c = A.contiguous();
    auto B_c = B.contiguous();

    const int N = static_cast<int>(A_c.size(0));
    const int K = static_cast<int>(A_c.size(1));
    const int M = static_cast<int>(B_c.size(1));

    auto C = torch::zeros({N, M}, A_c.options());

    const dim3 threads(TILE, TILE);
    const dim3 blocks((M + TILE - 1) / TILE, (N + TILE - 1) / TILE);

    const __hip_bfloat16* A_ptr =
        reinterpret_cast<const __hip_bfloat16*>(A_c.data_ptr<at::BFloat16>());
    const __hip_bfloat16* B_ptr =
        reinterpret_cast<const __hip_bfloat16*>(B_c.data_ptr<at::BFloat16>());
    __hip_bfloat16* C_ptr =
        reinterpret_cast<__hip_bfloat16*>(C.data_ptr<at::BFloat16>());

    hipLaunchKernelGGL(
        bf16_gemm_tiled_kernel,
        blocks,
        threads,
        0,
        0,
        A_ptr, B_ptr, C_ptr,
        N, K, M
    );

    auto err = hipGetLastError();
    TORCH_CHECK(err == hipSuccess, "bf16_gemm_tiled_kernel launch failed: ", hipGetErrorString(err));

    return C;
}
"""

_bf16_matmul_ext = load_inline(
    name="bf16_matmul_hip_ext",
    cpp_sources=_cpp_src,
    hip_sources=_hip_src,
    functions=["bf16_matmul_hip"],
    verbose=False,
    extra_cflags=["-O3"],
    extra_cuda_cflags=["-O3"],
    extra_ldflags=[],
)


class ModelNew(nn.Module):
    """
    Optimized model with a custom HIP BF16 tiled GEMM kernel for C = A @ B.
    """
    def __init__(self):
        super().__init__()
        self._ext = _bf16_matmul_ext

    def forward(self, A: torch.Tensor, B: torch.Tensor) -> torch.Tensor:
        if (
            A.is_cuda
            and B.is_cuda
            and A.dtype == torch.bfloat16
            and B.dtype == torch.bfloat16
            and A.dim() == 2
            and B.dim() == 2
        ):
            return self._ext.bf16_matmul_hip(A, B)
        return torch.matmul(A, B)
