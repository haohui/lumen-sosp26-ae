#include <torch/extension.h>
#include <hip/hip_runtime.h>
#include <cstdint>

constexpr int TILE = 16;

__device__ __forceinline__ float bf16_to_float(uint16_t x) {
    union {
        uint32_t u;
        float f;
    } v;
    v.u = static_cast<uint32_t>(x) << 16;
    return v.f;
}

__device__ __forceinline__ uint16_t float_to_bf16_rn(float x) {
    union {
        uint32_t u;
        float f;
    } v;
    v.f = x;
    uint32_t u = v.u;
    const uint32_t lsb = (u >> 16) & 1u;
    u += 0x7FFFu + lsb;  // round-to-nearest-even
    return static_cast<uint16_t>(u >> 16);
}

__global__ void bf16_gemm_tiled_kernel(
    const uint16_t* __restrict__ A,
    const uint16_t* __restrict__ B,
    uint16_t* __restrict__ C,
    int N, int K, int M
) {
    __shared__ uint16_t As[TILE][TILE];
    __shared__ uint16_t Bs[TILE][TILE];

    const int tx = threadIdx.x;
    const int ty = threadIdx.y;
    const int row = blockIdx.y * TILE + ty;
    const int col = blockIdx.x * TILE + tx;

    float acc = 0.0f;

    for (int t = 0; t < K; t += TILE) {
        const int a_col = t + tx;
        const int b_row = t + ty;

        As[ty][tx] = (row < N && a_col < K) ? A[row * K + a_col] : static_cast<uint16_t>(0);
        Bs[ty][tx] = (b_row < K && col < M) ? B[b_row * M + col] : static_cast<uint16_t>(0);

        __syncthreads();

    }
}
