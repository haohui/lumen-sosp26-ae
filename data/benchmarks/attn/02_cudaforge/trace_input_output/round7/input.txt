__device__ __forceinline__ float wave_reduce_sum_f32(float v) {
    const int lane = (int)(threadIdx.x % warpSize);
    for (int offset = (int)(warpSize >> 1); offset > 0; offset >>= 1) {
        float other = __shfl_down(v, offset);
        if (lane < offset) v += other;
    }
    return v;
}

__device__ __forceinline__ float wave_broadcast_lane0(float v) {
#if defined(__HIP_PLATFORM_AMD__)
    return __builtin_amdgcn_readfirstlane(v);
#else
    return __shfl(v, 0);
#endif
}

// Specialized D=64 kernel
__global__ __launch_bounds__(64, 8) void sdpa_causal_bf16_d64_kernel(
    const at::BFloat16* __restrict__ Q,
    const at::BFloat16* __restrict__ K,

    const long long kv_head_base = ((long long)b * Hk + kvh) * (long long)S * 64LL;

    float m = -INFINITY;
    float l = 0.0f;
    float acc = 0.0f;

    const at::BFloat16* k_ptr = K + kv_head_base;
    const at::BFloat16* v_ptr = V + kv_head_base;

    for (int j = 0; j <= i; ++j) {

}
